# Agent + Job Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a renderer-only AI agent (tailor resume, draft cover letters, chat coach) with per-job workspaces, wisp-style threaded chat, OS-keyring BYOK, readonly PDF job descriptions, and Flowbite settings UI.

**Architecture:** New `src/agent/` layer (threads, models, tools, keyring, JD store) beside `src/data/`; `Agent` from `@earendil-works/pi-agent-core` streams via `pi-ai`; mutations go only through 4 narrow tools into `workspace.ts` helpers; no backend, web + Electron parity.

**Tech Stack:** Vue 3 + TS strict, `@earendil-works/pi-ai`, `@earendil-works/pi-agent-core`, `pdf-parse` v2 (`PDFParse`), `flowbite` + `flowbite-vue`, Electron `safeStorage` IPC, IndexedDB (web JD bytes), vitest.

## Global Constraints

- Strict TypeScript: `pnpm typecheck` (`vue-tsc -b`) must pass after every task.
- Renderer imports are extensionless (no `.js` suffixes resolving to `.ts`).
- Workspace JSON never contains PDF bytes or API keys.
- JD text is readonly; only Replace-PDF updates it.
- Form view open in one pane only.
- TDD: failing test first, then minimal implementation, then commit per task.

```ts
export interface ChatMsg {
	id: string
	role: 'user' | 'assistant' | 'tool'
	text: string
	timestamp: number
}
export interface ChatThread {
	entryId: string | null
	edges: Record<string, string | null>
	messages: Record<string, ChatMsg>
	decisions: number[] | null
}

export function blankThread(): ChatThread {
	return { entryId: null, edges: {}, messages: {}, decisions: null }
}

export function addMessage(t: ChatThread, msg: ChatMsg, parentId?: string | null): void {
	t.messages[msg.id] = msg
	const parent = parentId === undefined ? (t.entryId === null ? null : getDefaultLeaf(t)) : parentId
	t.edges[msg.id] = parent
	if (t.entryId === null && parent === null) t.entryId = msg.id
}

export function getParent(t: ChatThread, id: string): string | null | undefined {
	return t.edges[id]
}

export function getChildren(t: ChatThread, parentId: string): string[] {
	return Object.keys(t.edges).filter((id) => t.edges[id] === parentId)
}

export function getPathTo(t: ChatThread, leafId: string): ChatMsg[] {
	const ids: string[] = []
	let cur: string | null | undefined = leafId
	while (typeof cur === 'string') {
		ids.push(cur)
		if (cur === t.entryId) break
		cur = t.edges[cur]
		if (cur === undefined) break
	}
	ids.reverse()
	return ids.map((id) => t.messages[id]).filter(Boolean)
}

export function getDefaultLeaf(t: ChatThread): string | null {
	if (!t.entryId) return null
	let cur = t.entryId
	for (;;) {
		const kids = getChildren(t, cur)
		if (!kids.length) return cur
		cur = kids[kids.length - 1]
	}
}

export function getTree(t: ChatThread): { key: string; parent: string | null; children: string[] }[] {
	return Object.keys(t.messages).map((id) => ({ key: id, parent: t.edges[id] ?? null, children: getChildren(t, id) }))
}
```

```ts
import { PDFParse } from 'pdf-parse'
import { isElectron } from '../data/persistence'

export async function extractJdText(pdf: ArrayBuffer): Promise<{ text: string; pageCount: number }> {
	const parser = new PDFParse({ data: new Uint8Array(pdf) } as unknown as { url: string })
	try {
		const info = await parser.getInfo({ parsePageInfo: true } as never)
		const res = await parser.getText()
		const pages = (info as { pages?: unknown[] }).pages
		return { text: (res.text as string).trim(), pageCount: Array.isArray(pages) ? pages.length : 0 }
	} finally {
		await parser.destroy()
	}
}

function idb(): Promise<IDBDatabase> {
	return new Promise((resolve, reject) => {
		const req = indexedDB.open('resume-tailor-jds', 1)
		req.onupgradeneeded = () => req.result.createObjectStore('pdfs')
		req.onsuccess = () => resolve(req.result)
		req.onerror = () => reject(req.error)
	})
}

export async function saveJdPdf(jobId: string, data: ArrayBuffer): Promise<string> {
	const pdfRefId = `${jobId}.pdf`
	if (isElectron() && window.electronAPI && 'invoke' in window.electronAPI) {
		await (window.electronAPI as unknown as { invoke: (c: string, ...a: unknown[]) => Promise<unknown> }).invoke(
			'agent-jd:save',
			{ jobId, data: Array.from(new Uint8Array(data)) },
		)
		return pdfRefId
	}
	const db = await idb()
	await new Promise<void>((resolve, reject) => {
		const tx = db.transaction('pdfs', 'readwrite')
		tx.objectStore('pdfs').put(data, pdfRefId)
		tx.oncomplete = () => resolve()
		tx.onerror = () => reject(tx.error)
	})
	return pdfRefId
}

export async function loadJdPdf(pdfRefId: string): Promise<ArrayBuffer | null> {
	if (isElectron() && window.electronAPI && 'invoke' in window.electronAPI) {
		const bytes = await (
			window.electronAPI as unknown as { invoke: (c: string, ...a: unknown[]) => Promise<number[] | null> }
		).invoke('agent-jd:load', pdfRefId)
		return bytes ? new Uint8Array(bytes).buffer : null
	}
	const db = await idb()
	return new Promise((resolve, reject) => {
		const tx = db.transaction('pdfs', 'readonly')
		const rq = tx.objectStore('pdfs').get(pdfRefId)
		rq.onsuccess = () => resolve((rq.result as ArrayBuffer) ?? null)
		rq.onerror = () => reject(rq.error)
	})
}

export function jdBlobUrl(data: ArrayBuffer, mime = 'application/pdf'): string {
	return URL.createObjectURL(new Blob([data], { type: mime }))
}
```

Note: `agent-jd:save/load` IPC handlers land in Task 5 with keyring IPC; until then Electron path throws at runtime only — web path is fully testable. Cap imports at 10MB / 20 pages at call sites (Task 7).

```ts
import { isElectron } from '../data/persistence'

const mem = new Map<string, string>()

type Bridge = { invoke: (channel: string, ...args: never[]) => Promise<unknown> }

function bridge(): Bridge | null {
	const api = window.electronAPI as unknown as { invoke?: Bridge['invoke'] } | undefined
	return typeof api?.invoke === 'function' ? { invoke: api.invoke } : null
}

export async function setApiKey(provider: string, key: string): Promise<void> {
	const b = bridge()
	if (b && isElectron()) {
		await b.invoke('agent-key:set', provider, key)
		return
	}
	mem.set(provider, key)
}

export async function getApiKey(provider: string): Promise<string | null> {
	const b = bridge()
	if (b && isElectron()) return (await b.invoke('agent-key:get', provider)) as string | null
	return mem.get(provider) ?? null
}

export async function deleteApiKey(provider: string): Promise<void> {
	const b = bridge()
	if (b && isElectron()) {
		await b.invoke('agent-key:delete', provider)
		return
	}
	mem.delete(provider)
}
```
